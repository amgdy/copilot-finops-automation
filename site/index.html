name: Copilot FinOps preview

on:
  workflow_dispatch: {}

permissions:
  contents: read

jobs:
  preview:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v7
      - name: Preview AI-credit budgets
        uses: amgdy/copilot-finops-automation@v3
        with:
          operation: apply
          config-file: config/copilot-finops.yml
          enterprise: ${{ vars.COPILOT_FINOPS_ENTERPRISE }}
          token: ${{ secrets.COPILOT_FINOPS_TOKEN }}
          dry-run: "true"
          log-level: info
